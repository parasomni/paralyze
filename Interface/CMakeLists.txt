cmake_minimum_required(VERSION 3.15)

include(${CMAKE_BINARY_DIR}/conan_toolchain.cmake)

# Declare the project immediately
project(Iface LANGUAGES CXX)

# Set C++ standard
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED True)

# Set the compiler
set(CMAKE_CXX_COMPILER clang++)
set(CMAKE_C_COMPILER clang)

# Specify source files
set(SOURCES
    Sources/ConfigWriter.cpp
    Sources/IfcEssentials.cpp
    Sources/IfcMain.cpp
)

# Add the executable
add_executable(Iface ${SOURCES})
target_include_directories(Iface PRIVATE ${CMAKE_SOURCE_DIR}/Headers)

if (CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(STATUS "Using Clang++")
    target_compile_options(Iface PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Use find_package to locate dependencies generated by Conan
find_package(fmt CONFIG REQUIRED)
find_package(mongocxx REQUIRED)

# Link dependencies to the target
target_link_libraries(Iface fmt::fmt-header-only mongo::mongocxx_shared)
