cmake_minimum_required(VERSION 3.15)

include(${CMAKE_BINARY_DIR}/conan_toolchain.cmake)

# Declare the project immediately
project(Tests LANGUAGES CXX)

# Set C++ standard
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED True)

# Set the compiler
set(CMAKE_CXX_COMPILER clang++)
set(CMAKE_C_COMPILER clang)

# Specify source files
set(SOURCES
    utest_ConfigWriter.cpp
    utest_ConfigModule.cpp
    ../Interface/Sources/IfcEssentials.cpp
    ../Interface/Sources/ConfigWriter.cpp
    ../Server/Sources/ConfigModule.cpp
    ../Server/Sources/Essentials.cpp
)

# Add the executable
add_executable(Utest ${SOURCES})
target_include_directories(Utest PRIVATE /home/xpr/repos/wizard/c++-implementation/Interface/Headers)
target_include_directories(Utest PRIVATE /home/xpr/repos/wizard/c++-implementation/Server/Headers)
target_include_directories(Utest PRIVATE /home/xpr/repos/wizard/c++-implementation/tests)

if (CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(STATUS "Using Clang++")
    target_compile_options(Utest PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Use find_package to locate dependencies generated by Conan
find_package(Boost CONFIG REQUIRED)
find_package(fmt CONFIG REQUIRED)
find_package(mongocxx REQUIRED)

# Link dependencies to the target
target_link_libraries(Utest Boost::boost fmt::fmt-header-only mongo::mongocxx_shared)
